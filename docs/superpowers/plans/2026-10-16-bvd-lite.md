```lua
-- Minimal headless test harness for BVD Lite pure modules (stock Lua 5.1).
local T = { passed = 0, failed = 0, current = "?" }
local ROOT = os.getenv("BVDL_LUA_ROOT")

function T.load(rel)
    return dofile(ROOT .. "/" .. rel)
end

local function fail(msg)
    T.failed = T.failed + 1
    print("  FAIL [" .. T.current .. "] " .. msg)
end

function T.eq(a, b, msg)
    if a ~= b then fail((msg or "eq") .. ": expected " .. tostring(b) .. ", got " .. tostring(a)) else T.passed = T.passed + 1 end
end

function T.near(a, b, tol, msg)
    if type(a) ~= "number" or math.abs(a - b) > tol then
        fail((msg or "near") .. ": expected " .. tostring(b) .. " ±" .. tostring(tol) .. ", got " .. tostring(a))
    else
        T.passed = T.passed + 1
    end
end

function T.truthy(v, msg)
    if not v then fail((msg or "truthy") .. ": got " .. tostring(v)) else T.passed = T.passed + 1 end
end

function T.case(name, fn)
    T.current = name
    local ok, err = pcall(fn)
    if not ok then fail("error: " .. tostring(err)) end
end

function T.finish()
    print(string.format("%d passed, %d failed", T.passed, T.failed))
    os.exit(T.failed == 0 and 0 or 1)
end

return T
```

```lua
-- BVDL_Shaper: mirror of vanilla 42.21 CarController's drive-force chain and its inverse.
-- Vanilla computes engineForce = enginePower * chain(state). Every chain term is
-- readable from Lua, and enginePower is writable (setEngineFeature), so Lite can make
-- Bullet receive any force F* by writing enginePower = F* / chain.
-- Pure math: no game API calls.

BVDL = BVDL or {}
BVDL.Shaper = BVDL.Shaper or {}
local M = BVDL.Shaper

M.MIN_CHAIN = 0.02
M.REDLINE_GUARD = 6900
M.CEILING_MULT = 4

-- CarController.update off-road limitMod (:310-322). Reverse counts as gear 1.
function M.offroadLimit(s)
    if not s.offroad then return 1.0 end
    local t = s.gear or 1
    if t < 1 then t = 1 end
    local clamped = math.min(math.max(t - 1, 0), 7)
    local mult = s.towing and 0.8 or 0.6
    return (1.0 - clamped / 15.0) * mult * (s.offroadEff or 1.0)
end

-- control_ForwardNew (:683-800). s.speed is signed km/h.
function M.forwardChain(s)
    if s.towing or s.gear < 1 then return nil end
    local c = (s.gear == 1) and 1.5 or 1.0
    c = c * (0.3 + s.rpm / 30000.0)
    c = c * (1.0 - s.speed / 200.0)
    if s.rpm > 6000 then
        c = c * ((7000.0 - s.rpm) / 1000.0)
    end
    local ms = s.maxSpeed
    if s.slow then
        c = c * 0.75
        if s.speed > ms * 0.6 then
            c = c * ((ms * 0.75 + 20.0 - s.speed) / 20.0)
        end
    end
    if s.fast then
        if s.speed > ms * 1.15 then
            c = c * ((ms * 1.15 + 20.0 - s.speed) / 20.0)
        end
    elseif s.speed > ms then
        c = c * ((ms + 20.0 - s.speed) / 20.0)
    end
    return c * M.offroadLimit(s)
end

-- control_Reverse (:796-826). Returns the magnitude (vanilla force is negative).
function M.reverseChain(s)
    if s.towing then return nil end
    local sp = s.speed * 1.5
    if sp < -s.maxSpeedReverse then return 0 end
    local c = 0.75 + s.rpm / 24000.0
    if s.rpm > 6000 then
        c = c * ((7000.0 - s.rpm) / 1000.0)
    end
    if s.slow then
        c = c * 0.7
        if sp < -5.0 then
            c = c * ((15.0 + sp) / 10.0)
        end
    end
    return c * M.offroadLimit({ offroad = s.offroad, gear = 1, towing = false, offroadEff = s.offroadEff })
end

-- Engine power that makes vanilla produce targetForce, or nil to hand back to vanilla.
function M.invert(targetForce, chain, basePower, rpm)
    if not chain or chain < M.MIN_CHAIN or rpm >= M.REDLINE_GUARD then
        return nil
    end
    local p = math.floor(targetForce / chain + 0.5)
    local ceiling = basePower * M.CEILING_MULT
    if p > ceiling then p = ceiling end
    if p < 1 then p = 1 end
    return p
end

return M
```

```lua
-- BVDL_Engine: torque-curve archetypes for BVD Lite. Curves are normalized
-- torque (peak 1.0) against normalized rpm (rpm / redline), piecewise linear.
-- Pure math: no game API calls.

BVDL = BVDL or {}
BVDL.Engine = BVDL.Engine or {}
local M = BVDL.Engine

M.ARCHETYPES = {
    diesel  = { redline = 4200, idle = 700,
        curve = { {0.15,0.62},{0.25,0.88},{0.35,1.00},{0.50,0.97},{0.65,0.88},{0.80,0.74},{0.92,0.58},{1.00,0.40} } },
    v8      = { redline = 5800, idle = 650,
        curve = { {0.15,0.70},{0.25,0.86},{0.35,0.96},{0.45,1.00},{0.60,0.98},{0.75,0.90},{0.90,0.76},{1.00,0.60} } },
    inline4 = { redline = 6800, idle = 800,
        curve = { {0.15,0.55},{0.30,0.72},{0.45,0.85},{0.60,0.95},{0.70,1.00},{0.82,0.96},{0.92,0.86},{1.00,0.70} } },
    sports  = { redline = 7200, idle = 900,
        curve = { {0.15,0.58},{0.30,0.76},{0.45,0.90},{0.55,0.97},{0.65,1.00},{0.80,0.97},{0.92,0.88},{1.00,0.74} } },
}

-- Placeholder until in-game calibration (plan Task 9 step "HP_PER_POWER").
M.HP_PER_POWER = 0.025

function M.normTorque(arch, x)
    if x > 1.0 then return 0 end
    local c = arch.curve
    if x <= c[1][1] then return c[1][2] end
    for i = 2, #c do
        local x1, y1 = c[i][1], c[i][2]
        if x <= x1 then
            local x0, y0 = c[i - 1][1], c[i - 1][2]
            return y0 + (y1 - y0) * (x - x0) / (x1 - x0)
        end
    end
    return c[#c][2]
end

function M.torque(arch, rpm, peakTorque)
    return M.normTorque(arch, rpm / arch.redline) * peakTorque
end

-- Peak torque (N·m) such that max(torque * omega) equals hp.
function M.peakTorqueFromHp(arch, hp)
    local best = 0
    for i = 1, 100 do
        local x = i / 100
        local v = M.normTorque(arch, x) * x
        if v > best then best = v end
    end
    local omegaAtRedline = arch.redline * 2 * math.pi / 60
    return hp * 745.7 / (best * omegaAtRedline)
end

local function has(name, list)
    for i = 1, #list do
        if string.find(name, list[i], 1, true) then return true end
    end
    return false
end

-- Order matters: "pickuptruck" must match the pickup rule before "truck".
local RULES = {
    { "sports",  { "sports" } },
    { "v8",      { "pickup", "suv", "offroad", "luxury", "stationwagon", "lights", "police", "sheriff" } },
    { "diesel",  { "stepvan", "ambulance", "truck", "bus", "van" } },
}

function M.select(scriptName, override)
    if type(override) == "string" and M.ARCHETYPES[override] then return override end
    if type(scriptName) ~= "string" then return "inline4" end
    local n = string.lower(scriptName)
    for i = 1, #RULES do
        if has(n, RULES[i][2]) then return RULES[i][1] end
    end
    return "inline4"
end

function M.vehicleClass(scriptName)
    if type(scriptName) ~= "string" then return "car" end
    local n = string.lower(scriptName)
```

```lua
-- BVDL_Gearbox: virtual gearbox for BVD Lite. Vanilla's own gear/rpm are ignored;
-- Bullet only ever sees the final drive force, so gears live entirely here.
-- Default ratio sets match BVD Full's DEFAULT_GEARBOX_3/4/5 (reverse first).
-- Pure math: no game API calls.

BVDL = BVDL or {}
BVDL.Gearbox = BVDL.Gearbox or {}
local M = BVDL.Gearbox

M.FINAL_DRIVE = 3.7
M.UP = 0.85
M.DOWN = 0.45
M.KICK = 0.55
M.KICK_HOLD = 0.4
M.SHIFT_TIME = 0.35
M.SHIFT_CUT = 0.2
M.LOCKUP = 2000
M.BAND = 800

M.SETS = {
    [3] = { r = 2.6, 2.6, 1.6, 1.0 },
    [4] = { r = 3.0, 3.0, 1.8, 1.3, 1.0 },
    [5] = { r = 3.2, 3.2, 2.0, 1.5, 1.15, 0.9 },
}

local function copySet(src)
    local out = { r = src.r }
    for i = 1, #src do out[i] = src[i] end
    return out
end

local function validOverride(o)
    if type(o) ~= "table" or #o < 4 or #o > 9 then return false end
    for i = 1, #o do
        if type(o[i]) ~= "number" or o[i] <= 0 or o[i] ~= o[i] then return false end
    end
    return true
end

function M.ratiosFor(count, override)
    if validOverride(override) then
        local out = { r = override[1] }
        for i = 2, #override do out[i - 1] = override[i] end
        return out
    end
    local n = tonumber(count) or 4
    if n < 3 then n = 3 end
    if n > 5 then n = 5 end
    return copySet(M.SETS[n])
end

function M.new(set)
    return { ratios = set, gear = 1, shiftLeft = 0, kickHeld = 0 }
end

function M.wheelRpm(speedKmh, radius)
    local v = math.abs(speedKmh) / 3.6
    return v / (2 * math.pi * radius) * 60
end

function M.ratio(st)
    if st.gear < 0 then return st.ratios.r * M.FINAL_DRIVE end
    return st.ratios[st.gear] * M.FINAL_DRIVE
end

function M.driveScale(st)
    if st.shiftLeft > 0 then return M.SHIFT_CUT end
    return 1
end

local function shiftTo(st, g)
    st.gear = g
    st.shiftLeft = M.SHIFT_TIME
    st.kickHeld = 0
end

function M.step(st, inp, dt)
    local w = M.wheelRpm(inp.speedKmh, inp.radius)
    local set = st.ratios
    if inp.reverse then
        st.gear = -1
        st.shiftLeft = 0
        st.kickHeld = 0
        return w * set.r * M.FINAL_DRIVE
    end
    if st.gear < 1 then st.gear = 1 end
    local n = #set
    if st.shiftLeft > 0 then
        st.shiftLeft = math.max(0, st.shiftLeft - dt)
    end
    local function rpmIn(g) return w * set[g] * M.FINAL_DRIVE end

    if inp.manual then
        local d = inp.manualDelta or 0
        if d ~= 0 then
            local g = math.min(math.max(st.gear + d, 1), n)
            if g ~= st.gear then shiftTo(st, g) end
        end
        return rpmIn(st.gear)
    end

    local rl = inp.redline
    if st.shiftLeft == 0 then
        if st.gear < n and rpmIn(st.gear) >= M.UP * rl then
            shiftTo(st, st.gear + 1)
        elseif st.gear > 1 and rpmIn(st.gear) < M.DOWN * rl and rpmIn(st.gear - 1) < M.UP * rl then
            shiftTo(st, st.gear - 1)
        end
    end

    if inp.throttle and st.gear > 1 and rpmIn(st.gear) < M.KICK * rl
        and rpmIn(st.gear - 1) < M.UP * rl * 0.98 then
        st.kickHeld = st.kickHeld + dt
    else
        st.kickHeld = 0
    end
    if st.kickHeld >= M.KICK_HOLD and st.shiftLeft == 0 then
        shiftTo(st, st.gear - 1)
    end
    return rpmIn(st.gear)
end

-- Torque-converter slip: below lockup the engine can sit above wheel-derived rpm,
-- and torque is multiplied (LowSpeedGrunt knob, default 2.5 -> up to 1.6x).
function M.converter(rpm, throttle, grunt, idle)
    local floor = throttle and (M.LOCKUP - M.BAND) or idle
    local eff = math.max(rpm, floor)
    if rpm >= M.LOCKUP then return eff, 1 end
    local slip = (M.LOCKUP - rpm) / M.LOCKUP
    if slip < 0 then slip = 0 end
    if slip > 1 then slip = 1 end
    local g = grunt or 2.5
    return eff, 1 + (g - 1) * 0.4 * slip
end

return M
```

```lua
-- BVDL_Drag: aerodynamic + rolling resistance for BVD Lite. Vanilla 42.21 has almost
-- no speed-dependent drag on paved road (probe 2026-10-16), so Lite supplies all of it.
-- Pure math: no game API calls.

BVDL = BVDL or {}
BVDL.Drag = BVDL.Drag or {}
local M = BVDL.Drag

M.CDA = { car = 0.70, van = 1.05, truck = 1.60 }
M.RHO = 1.2
M.CRR = 0.013
M.G = 9.81
M.CREEP_V = 0.3   -- m/s; rolling resistance fades to 0 below this so a parked car never creeps

function M.force(vMs, massKg, class, cfg, offroad)
    local v = math.abs(vMs)
    local cda = M.CDA[class] or M.CDA.car
    local aero = 0.5 * M.RHO * cda * v * v * (cfg.Drag or 1.0)
    local rr = offroad and (cfg.RollResistanceOffroad or 1.8) or (cfg.RollResistance or 1.0)
    local roll = M.CRR * massKg * M.G * rr
    if v < M.CREEP_V then roll = roll * v / M.CREEP_V end
    return aero + roll
end

-- Bisection on f(v) = force(v) * v - P over 0..150 m/s.
function M.topSpeed(wheelPowerW, massKg, class, cfg)
    local lo, hi = 0.0, 150.0
    for _ = 1, 60 do
        local mid = (lo + hi) / 2
        if M.force(mid, massKg, class, cfg, false) * mid > wheelPowerW then hi = mid else lo = mid end
    end
    return (lo + hi) / 2
end

return M
```

```lua
-- friction by its own frictionMul (BaseVehicle.java:9336-9353, 0.7 offroad, -0.3 rain)
-- before sending it to Bullet; Lite divides that back out so BVD's own surface model

local M = BVDL.Grip

M.CG = { car = 0.55, van = 0.75, truck = 0.95 }

M.G = 9.81

    if raining then m = m - 0.3 end
    return m
end

function M.family(fullType)
    if type(fullType) ~= "string" then return nil end
    local name = string.match(fullType, "%.([^%.]+)$") or fullType
    return (string.gsub(name, "%d+$", ""))
end

local function p(profile, k)
    local v = profile and profile[k]
    if type(v) == "number" and v > 0 then return v end
    return 1.0
end

function M.surfaceMul(cfg, profile, surf)
    local g = (cfg.GripLevel or 1.0) * p(profile, "road")
    if surf.offroad then
        g = g * (cfg.OffroadGrip or 0.85) * p(profile, "offroad") / p(profile, "road")
        local floor = (cfg.OffroadFloor or 0.55) * (cfg.GripLevel or 1.0)
        if g < floor then g = floor end
    end
    local wet = surf.wet or 0
    if wet > 0 then
        g = g * (1 - wet * (1 - (cfg.WetGrip or 0.7) * p(profile, "wet")))
    end
    local snow = surf.snow or 0
    if snow > 0 then
        g = g * (1 - snow * (1 - (cfg.SnowGrip or 0.45) * p(profile, "snow")))
    end
    return g
end

-- Longitudinal load transfer relative to a static 50/50 split.
function M.transferMul(isFront, accelMs2, cgHeight, wheelbase)
    if not wheelbase or wheelbase <= 0 then return 1.0 end
    local d = accelMs2 * cgHeight / (wheelbase * M.G)
    local ratio = isFront and (1 - d) or (1 + d)
    if ratio <= 0 then return 0.5 end
    local m = ratio ^ 0.8
    if m < 0.5 then m = 0.5 end
    if m > 1.4 then m = 1.4 end
    return m
end

function M.partFriction(base, surfaceMul, transferMul, vanillaMul)
    if type(base) ~= "number" or base <= 0 then return nil end
    local vm = vanillaMul
    if vm < 0.1 then vm = 0.1 end
    return base * surfaceMul * transferMul / vm
end

function M.shouldWrite(last, new)
    if last == nil then return true end
    return math.abs(new - last) > 0.01 * math.abs(last)
end

return M
```

```lua
-- BVDL_Drive: one Lite tick as pure math. Inputs are plain numbers gathered by
-- BVDL_Runtime; outputs are the accelerations Runtime converts to PZ units.
-- Pure math: no game API calls.

BVDL = BVDL or {}
BVDL.Drive = BVDL.Drive or {}
local M = BVDL.Drive
local E, G, Dr, Gr = BVDL.Engine, BVDL.Gearbox, BVDL.Drag, BVDL.Grip

M.ETA = 0.85   -- drivetrain efficiency
M.EB = 0.35    -- engine-braking torque as a fraction of peak torque at redline
M.DEFAULT_MASS = 1300

function M.vehicle(spec)
    local arch = E.select(spec.scriptName, spec.archOverride)
    local archDef = E.ARCHETYPES[arch]
    local class = E.vehicleClass(spec.scriptName)
    local hp = E.hpFor(spec.hp, nil)
    local mass = (type(spec.massReal) == "number" and spec.massReal > 0) and spec.massReal or M.DEFAULT_MASS
    return {
        arch = arch, archDef = archDef, class = class,
        peakTorque = E.peakTorqueFromHp(archDef, hp),
        radius = Gr.radius(class),
        massReal = mass,
        set = G.ratiosFor(spec.gearCount, spec.gearOverride),
    }
end

function M.compute(veh, st, cfg, f)
    local a = veh.archDef
    local rpmW = G.step(st, {
        speedKmh = f.kmh, radius = veh.radius, redline = a.redline,
        throttle = f.throttle, reverse = f.reverse, manual = f.manual, manualDelta = f.manualDelta,
    }, f.dt)
    local rpm, tmul = G.converter(rpmW, f.throttle, cfg.LowSpeedGrunt, a.idle)
    local tq = E.torque(a, rpm, veh.peakTorque) * tmul * (cfg.EnginePower or 1.0)
    local ratio = G.ratio(st)

    local drive = 0
    if f.throttle and rpmW < a.redline then
        drive = tq * ratio * M.ETA / veh.radius * G.driveScale(st)
    end
    if f.reverse and math.abs(f.kmh) >= (cfg.ReverseTopSpeed or 25) then
        drive = 0
    end

    local laden = 1
    if cfg.LoadAffectsHandling and type(f.ladenRatio) == "number" and f.ladenRatio > 1 then
        laden = f.ladenRatio
    end
    local m = veh.massReal * laden
    local drag = Dr.force(f.kmh / 3.6, m, veh.class, cfg, f.offroad)

    local eb = 0
    if not f.throttle and st.gear >= 1 then
        eb = M.EB * veh.peakTorque * math.min(rpmW / a.redline, 1) * ratio / veh.radius
    end

    local net = drive - drag
    local driveAccel, resistAccel
    if f.throttle then
        driveAccel = math.max(net, 0) / m
        resistAccel = math.max(-net, 0) / m
    else
        driveAccel = 0
        resistAccel = (drag + eb) / m
    end
    if resistAccel ~= resistAccel then resistAccel = 0 end
    if driveAccel ~= driveAccel then driveAccel = 0 end
    return {
        rpm = rpm, gear = st.gear, torque = tq, driveN = drive, dragN = drag, ebN = eb,
        massEff = m, driveAccel = driveAccel, resistAccel = resistAccel,
    }
end

return M
```

```lua
        while list[1] and now - list[1] > R.FOREIGN_WINDOW_MS do table.remove(list, 1) end
        if #list >= R.FOREIGN_LIMIT then
            kill(vehicle, "engine power keeps being changed by something else")
            return false
        end
        -- Legitimate change (e.g. engine damage): adopt as the new original.
        o.power = p
        logOnce("rebase:" .. tostring(vehicle:getScriptName()), "engine power changed externally; re-based")
    end
    return true
end

local function applyGrip(vehicle, cfg, kmh)
    local raining = RainManager.isRaining() and true or false
    local cm = getClimateManager()
    local sq = vehicle:getSquare()
    local outside = sq and sq:isOutside()
    local surf = {
        offroad = vehicle:isDoingOffroad(),
        wet = (outside and cm) and cm:getRainIntensity() or 0,
        snow = (outside and cm) and cm:getSnowIntensity() or 0,
    }
    local vm = Gr.vanillaMul(surf.offroad, math.abs(kmh), raining)
    local o = vehicle:getModData()[ORIG]
    local comp = {}
    for _, w in ipairs(cur.wheels) do
        local part = vehicle:getPartById(w.partId)
        local item = part and part:getInventoryItem()
        if part and item then
            local last = cur.lastFriction[w.partId]
            local now = part:getWheelFriction()
            -- updatePartStats rewrote it (tyre change/repair): adopt as the new base.
            if last == nil or math.abs(now - last) > 1e-4 then
                o.friction[w.partId] = now
            end
            local base = o.friction[w.partId]
            local family = Gr.family(item:getFullType())
            local profile = (cfg.TireGripModel and BVD.getTireProfile) and BVD.getTireProfile(family) or nil
            local sm = Gr.surfaceMul(cfg, profile, surf)
            local tm = Gr.transferMul(w.front, cur.accel, Gr.cgHeight(cur.veh.class), cur.wheelbase)
            local f = Gr.partFriction(base, sm, tm, vm)
            if f and Gr.shouldWrite(last, f) then
                part:setWheelFriction(f)
                cur.lastFriction[w.partId] = f
            end
            comp.tireFamily = family
            comp.gripRoad = Gr.surfaceMul(cfg, profile, { offroad = false, wet = 0, snow = 0 })
            comp.gripWet = Gr.surfaceMul(cfg, profile, { offroad = false, wet = 1, snow = 0 })
            comp.gripSnow = Gr.surfaceMul(cfg, profile, { offroad = false, wet = 0, snow = 1 })
            comp.gripOffroad = Gr.surfaceMul(cfg, profile, { offroad = true, wet = 0, snow = 0 })
        end
    end
    return comp
end

local function applyResistance(vehicle, decel)
    if decel <= 0 then return end
    if not tmpVec then tmpVec = Vector3f.new() end
    local v = vehicle:getLinearVelocity(tmpVec)
    local vx, vz = v:get(0), v:get(2)
    local speed = math.sqrt(vx * vx + vz * vz)
    if speed < 0.5 then return end
    local strength = decel * vehicle:getMass() * R.CAL.IMPULSE
    vehicle:addImpulse(Vector3f.new(-vx / speed * strength, 0, -vz / speed * strength), Vector3f.new(0, 0, 0))
end

local function tickVehicle(player, vehicle)
    if cur == nil or cur.vehicle ~= vehicle then
        if cur then restore(cur.vehicle, "switched vehicle") end
        begin(vehicle)
    end
    local o = vehicle:getModData()[ORIG]
    if not o then ledgerOpen(vehicle) o = vehicle:getModData()[ORIG] end
    if not checkForeign(vehicle, o) then return end

    local cfg = BVD.cfg()
    local now = getTimestampMs()
    local dt = (now - cur.lastMs) / 1000
    if dt <= 0 then dt = 0.016 end
    if dt > 0.1 then dt = 0.1 end
    local kmh = vehicle:getCurrentSpeedKmHour()
    local a = ((kmh - cur.lastKmh) / 3.6) / dt
    cur.accel = cur.accel * 0.8 + a * 0.2
    cur.lastKmh, cur.lastMs = kmh, now

    local throttle, reverse, keyboard = throttleIntent(vehicle)
    local fg = BetterVehicleDynamicsMod.forceGear or 0
    local delta = fg - cur.lastForceGear
    cur.lastForceGear = fg

    local initial = vehicle.getInitialMass and vehicle:getInitialMass() or vehicle:getMass()
    local laden = (initial and initial > 0) and (vehicle:getMass() / initial) or 1
    local out = D.compute(cur.veh, cur.st, cfg, {
        kmh = kmh, throttle = throttle or reverse, reverse = reverse, offroad = vehicle:isDoingOffroad(), dt = dt,
        manual = BetterVehicleDynamicsMod.manualShift == true, manualDelta = delta, ladenRatio = laden,
    })

    -- Drive force through the vanilla chain.
    local targetForce = out.driveAccel * vehicle:getMass() * R.CAL.FORCE
    local state = {
        gear = vehicle:getTransmissionNumber(), rpm = vehicle:getEngineSpeed(), speed = kmh,
        maxSpeed = vehicle:getMaxSpeed(), maxSpeedReverse = vehicle:getScript():getMaxSpeedReverse(),
        offroad = vehicle:isDoingOffroad(), offroadEff = vehicle:getScript():getOffroadEfficiency(),
        slow = player:hasTrait(CharacterTrait.SUNDAY_DRIVER), fast = player:hasTrait(CharacterTrait.SPEED_DEMON),
        towing = vehicle:getVehicleTowing() ~= nil or vehicle:getVehicleTowedBy() ~= nil,
    }
    local chain
    if reverse then chain = S.reverseChain(state) else chain = S.forwardChain(state) end
    local power = S.invert(targetForce, chain, o.power, state.rpm)
    if power == nil then power = o.power end
    if power ~= vehicle:getEnginePower() then setPower(vehicle, power) end
    cur.lastPower = power

    if keyboard then applyResistance(vehicle, out.resistAccel) end

    cur.tick = cur.tick + 1
    local comp = cur.comp or {}
    if cur.tick % R.GRIP_EVERY == 0 then comp = applyGrip(vehicle, cfg, kmh) cur.comp = comp end

    comp.mode = "Lite"
    comp.gear = out.gear
    comp.rpm = out.rpm
    comp.torque = out.torque
    comp.engineForce = targetForce
    comp.drag = out.dragN
    comp.archetype = cur.veh.arch
    comp.ladenRatio = laden
    BetterVehicleDynamicsMod.computed = comp
end

local function onTick()
    if R.decideMode() ~= "Lite" then return end
    local player = getPlayer()
    local vehicle = player and player:getVehicle()
    if cur and cur.vehicle ~= vehicle then restore(cur.vehicle, "left vehicle") end
    if not vehicle or not vehicle:isDriver(player) or not vehicle:isEngineRunning() or killed[vehicle] then
        return
    end
    local ok, err = pcall(tickVehicle, player, vehicle)
    if not ok then
        kill(vehicle, "error: " .. tostring(err))
    end
end

local function onSave()
    if cur then restore(cur.vehicle, "save") end
end

local function onEnterVehicle(character)
    if character ~= getPlayer() then return end
    R.decideMode()
    local vehicle = character:getVehicle()
    -- Repair a ledger left behind by a crash/quit while nobody is shaping it.
    if vehicle and vehicle:getModData()[ORIG] and (cur == nil or cur.vehicle ~= vehicle) then
        restore(vehicle, "stale ledger")
    end
end

local function onPlayerDeath(p)
    if p == getPlayer() and cur then restore(cur.vehicle, "death") end
end

Events.OnTick.Add(onTick)
Events.OnSave.Add(onSave)
Events.OnEnterVehicle.Add(onEnterVehicle)
Events.OnPlayerDeath.Add(onPlayerDeath)

return R
```

```lua
-- BVD Lite Bench: logs one line per 250 ms while driving, plus scenario markers.
-- F10 cycles scenarios: IDLE -> COAST (coast-down, no impulses) -> IMPULSE (coast with
-- a fixed extra impulse) -> IDLE. Lines tagged [BVDL-BENCH] in console.txt.
local SCEN = { "IDLE", "COAST", "IMPULSE" }
local scen = 1
local lastMs, lastKmh = 0, nil
local IMPULSE_STRENGTH = 2000
local tmp = nil

local function log(msg) print("[BVDL-BENCH] " .. msg) end

local function onKey(key)
    if key ~= Keyboard.KEY_F10 then return end
    scen = scen % #SCEN + 1
    local p = getPlayer()
    if p then p:Say("Bench: " .. SCEN[scen]) end
    local v = p and p:getVehicle()
    if v then
        log(string.format("scenario %s script %s mass %.1f initialMass %.1f power %.0f",
            SCEN[scen], tostring(v:getScriptName()), v:getMass(), v:getInitialMass(), v:getEnginePower()))
    end
end

local function onTick()
    local p = getPlayer()
    local v = p and p:getVehicle()
    if not v or not v:isDriver(p) then return end
    if SCEN[scen] == "IMPULSE" then
        if not tmp then tmp = Vector3f.new() end
        local lv = v:getLinearVelocity(tmp)
        local vx, vz = lv:get(0), lv:get(2)
        local sp = math.sqrt(vx * vx + vz * vz)
        if sp > 0.5 then
            v:addImpulse(Vector3f.new(-vx / sp * IMPULSE_STRENGTH, 0, -vz / sp * IMPULSE_STRENGTH), Vector3f.new(0, 0, 0))
        end
    end
    local now = getTimestampMs()
    if now - lastMs < 250 then return end
    local kmh = v:getCurrentSpeedKmHour()
    local accel = lastKmh and ((kmh - lastKmh) / ((now - lastMs) / 1000)) or 0
    lastMs, lastKmh = now, kmh
    local c = BetterVehicleDynamicsMod and BetterVehicleDynamicsMod.computed or {}
    log(string.format("%s kmh %.1f accel %.2f mass %.1f power %.0f lgear %s lrpm %.0f force %.0f drag %.0f mode %s",
        SCEN[scen], kmh, accel, v:getMass(), v:getEnginePower(), tostring(c.gear), c.rpm or -1,
        c.engineForce or -1, c.drag or -1, tostring(c.mode)))
end

Events.OnKeyPressed.Add(onKey)
Events.OnTick.Add(onTick)
log("loaded; F10 cycles IDLE/COAST/IMPULSE")
```
